import { Router } from "express";
import mongoose from "mongoose";
import Ward from "../models/Ward.js";
import Room from "../models/Room.js";
import Bed from "../models/Bed.js";
import Department from "../models/Department.js";
import Admission from "../models/Admission.js";
import { authenticate, allowRoles } from "../middleware/auth.js";
const router = Router();
router.use(authenticate, allowRoles("Admin", "Receptionist"));
const idOK = mongoose.isValidObjectId;
const bedStatuses = ["Available", "Reserved", "Occupied", "Cleaning", "Under Maintenance"];
router.get("/wards", async (_req,res,next)=>{try{res.json(await Ward.find({active:true}).populate("department","name").sort({name:1}));}catch(e){next(e);}});
router.get("/rooms", async (_req,res,next)=>{try{res.json(await Room.find({active:true}).populate({path:"ward",populate:{path:"department",select:"name"}}).sort({roomNumber:1}));}catch(e){next(e);}});
router.get("/beds", async (_req,res,next)=>{try{const rows=await Bed.find().populate({path:"room",populate:{path:"ward",populate:{path:"department",select:"name"}}}).sort({bedNumber:1});res.json(rows.filter(b=>b.room?.active!==false&&b.room?.ward?.active!==false));}catch(e){next(e);}});
router.post("/wards", async (req,res,next)=>{try{const {name,department,wardType="General",capacity}=req.body;if(typeof name!=="string"||!name.trim()||!idOK(department)||!Number.isInteger(Number(capacity))||Number(capacity)<1||Number(capacity)>500)return res.status(400).json({message:"Ward name, valid department and capacity (1–500) are required"});if(!(await Department.exists({_id:department,active:true})))return res.status(404).json({message:"Active department not found"});const ward=await Ward.create({name:name.trim(),department,wardType,capacity:Number(capacity)});res.status(201).json(await ward.populate("department","name"));}catch(e){if(e.code===11000)return res.status(409).json({message:"This ward already exists in the department"});next(e);}});
router.patch("/wards/:id", async (req,res,next)=>{try{if(!idOK(req.params.id))return res.status(400).json({message:"Invalid ward ID"});const updates={};for(const k of ["name","wardType"])if(k in req.body)updates[k]=req.body[k];if("capacity" in req.body){const n=Number(req.body.capacity);if(!Number.isInteger(n)||n<1||n>500)return res.status(400).json({message:"Capacity must be from 1 to 500"});updates.capacity=n;}const current=await Ward.findById(req.params.id);if(!current)return res.status(404).json({message:"Ward not found"});if(updates.capacity){const rooms=await Room.find({ward:current._id,active:true}).select("_id");const bedCount=await Bed.countDocuments({room:{$in:rooms.map(r=>r._id)}});if(updates.capacity<bedCount)return res.status(409).json({message:`Capacity cannot be below its ${bedCount} configured beds`});}Object.assign(current,updates);await current.save();const ward=await current.populate("department","name");res.json(ward);}catch(e){next(e);}});
router.delete("/wards/:id", async (req,res,next)=>{try{if(!idOK(req.params.id))return res.status(400).json({message:"Invalid ward ID"});const rooms=await Room.find({ward:req.params.id,active:true});if(rooms.length)return res.status(409).json({message:"Remove or deactivate this ward's rooms first"});const ward=await Ward.findByIdAndUpdate(req.params.id,{active:false},{new:true});if(!ward)return res.status(404).json({message:"Ward not found"});res.json({message:"Ward deactivated"});}catch(e){next(e);}});
router.post("/rooms", async (req,res,next)=>{try{const {ward,roomNumber,roomType="Standard",bedCount}=req.body;const count=Number(bedCount);const dailyRate=Number(req.body.dailyRate ?? 1200);if(!idOK(ward)||typeof roomNumber!=="string"||!roomNumber.trim()||!Number.isInteger(count)||count<1||count>20||!Number.isFinite(dailyRate)||dailyRate<0||dailyRate>100000)return res.status(400).json({message:"Choose a ward, room number, bed count (1–20) and valid daily room rate"});const w=await Ward.findOne({_id:ward,active:true});if(!w)return res.status(404).json({message:"Active ward not found"});const currentBeds=await Bed.countDocuments({room:{$in:(await Room.find({ward,active:true}).select("_id")).map(r=>r._id)}});if(currentBeds+count>w.capacity)return res.status(409).json({message:`Ward capacity is ${w.capacity}; ${currentBeds} beds are already configured`});const room=await Room.create({ward,roomNumber:roomNumber.trim(),roomType,bedCount:count,dailyRate});await Bed.insertMany(Array.from({length:count},(_,i)=>({room:room._id,bedNumber:`${roomNumber.trim()}-${i+1}`})));res.status(201).json(await room.populate({path:"ward",populate:{path:"department",select:"name"}}));}catch(e){if(e.code===11000)return res.status(409).json({message:"Room or bed numbers already exist"});next(e);}});
router.patch("/rooms/:id", async (req,res,next)=>{try{if(!idOK(req.params.id))return res.status(400).json({message:"Invalid room ID"});const room=await Room.findById(req.params.id);if(!room)return res.status(404).json({message:"Room not found"});if("roomType" in req.body)room.roomType=req.body.roomType;if("dailyRate" in req.body){const rate=Number(req.body.dailyRate);if(!Number.isFinite(rate)||rate<0||rate>100000)return res.status(400).json({message:"Daily rate must be between 0 and 100000"});room.dailyRate=rate;}if("roomNumber" in req.body){if(typeof req.body.roomNumber!=="string"||!req.body.roomNumber.trim())return res.status(400).json({message:"Room number is required"});room.roomNumber=req.body.roomNumber.trim();}await room.save();res.json(room);}catch(e){if(e.code===11000)return res.status(409).json({message:"Room number already exists in this ward"});next(e);}});
router.delete("/rooms/:id", async(req,res,next)=>{try{if(!idOK(req.params.id))return res.status(400).json({message:"Invalid room ID"});const beds=await Bed.find({room:req.params.id});if(beds.some(b=>b.status!=="Available"))return res.status(409).json({message:"Room has beds that are not available"});await Bed.updateMany({room:req.params.id},{$set:{status:"Under Maintenance"}});const room=await Room.findByIdAndUpdate(req.params.id,{active:false},{new:true});if(!room)return res.status(404).json({message:"Room not found"});res.json({message:"Room and its available bed records removed"});}catch(e){next(e);}});
router.post("/beds", async(req,res,next)=>{try{const {room,bedNumber,notes=""}=req.body;if(!idOK(room)||typeof bedNumber!=="string"||!bedNumber.trim())return res.status(400).json({message:"Room and bed number are required"});if(!(await Room.exists({_id:room,active:true})))return res.status(404).json({message:"Active room not found"});const bed=await Bed.create({room,bedNumber:bedNumber.trim(),notes});res.status(201).json(await bed.populate("room"));}catch(e){if(e.code===11000)return res.status(409).json({message:"That bed number already exists in this room"});next(e);}});
router.patch("/beds/:id/status", async(req,res,next)=>{try{if(!idOK(req.params.id))return res.status(400).json({message:"Invalid bed ID"});if(!bedStatuses.includes(req.body.status))return res.status(400).json({message:"Invalid bed status"});const bed=await Bed.findById(req.params.id);if(!bed)return res.status(404).json({message:"Bed not found"});if(bed.status==="Occupied"||await Admission.exists({bed:bed._id,status:"Admitted"}))return res.status(409).json({message:"An occupied bed can only be released by discharging or transferring the patient"});if(req.body.status==="Occupied")return res.status(400).json({message:"Use the admission workflow to occupy a bed"});bed.status=req.body.status;bed.notes=typeof req.body.notes==="string"?req.body.notes.slice(0,300):bed.notes;await bed.save();res.json(bed);}catch(e){next(e);}});
router.get("/availability",async(_req,res,next)=>{try{const [all,byStatus]=await Promise.all([Bed.countDocuments(),Bed.aggregate([{$group:{_id:"$status",count:{$sum:1}}}])]);res.json({total:all,byStatus:Object.fromEntries(byStatus.map(x=>[x._id,x.count]))});}catch(e){next(e);}});
export default router;
